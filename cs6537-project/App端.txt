const cleanMarkdown = (raw : string) : string => {
	return raw.replace(/[#*`]/g, '').trim();
};

const handleOptimize = () => {
	if (originalText.value == "") return uni.showToast({title:'内容不能为空', icon:'none'});
	const config = API_CONFIG_MAP.get(selectedModelId.value);
	if (config == null) return;
	loading.value = true;
	optimizedText.value = '';
	let systemMsg = "你是一个顶级电商文案专家。";
	if (selectedModelId.value === 'deepseek') {
		systemMsg += "你的任务是将简短的商品信息扩写为专业的 FAQ 问答。要求：针对消费者的潜在顾虑脑补并设计 4-5 组深度对答。严禁 Markdown 符号。";
	} else if (selectedModelId.value === 'kimi') {
		systemMsg += "你的任务是将商品信息扩写为极具种草力的小红书风格列表。要求：增加大量描述词、Emoji，内容长度增加至原有的 3 倍左右。严禁 Markdown 符号。";
	} else {
		systemMsg += "你的任务是将商品信息扩写为高端、具有品牌质感的深度营销叙述。要求：运用修辞增加对细节、体验的描绘，形成两段长度适中的专业段落。严禁 Markdown 符号。";
	}
	uni.request({
		url: config.get("baseUrl") as string,
		method: 'POST',
		header: { 'Authorization': 'Bearer ' + (config.get("apiKey") as string).trim(), 'Content-Type': 'application/json' },
		data: { "model": config.get("model"), "messages": [ { "role": "system", "content": systemMsg }, { "role": "user", "content": "请对以下内容执行 GEO 优化：" + originalText.value } ], "temperature": 0.8 },
		success: (res) => {
			if (res.statusCode == 200) {
				const resData = res.data as UTSJSONObject;
				const choices = resData.get('choices') as any[];
				const content = ((choices[0] as UTSJSONObject).get('message') as UTSJSONObject).get('content') as string;
				optimizedText.value = cleanMarkdown(content);
			}
		},
		complete: () => { loading.value = false; }
	});
};

const handleSimulate = () => {
	if (optimizedText.value == "") return uni.showToast({title:'请先优化', icon:'none'});
	const config = API_CONFIG_MAP.get(selectedModelId.value); 
	if (config == null) return;
	simLoading.value = true;
	simulationResult.value = '';
	uni.request({
		url: config.get("baseUrl") as string,
		method: 'POST',
		header: { 'Authorization': 'Bearer ' + (config.get("apiKey") as string).trim(), 'Content-Type': 'application/json' },
		data: { "model": config.get("model"), "messages": [ { "role": "system", "content": (activeSim.value.get('prompt') as string) }, { "role": "user", "content": optimizedText.value } ], "temperature": 0.3 },
		success: (res) => {
			if (res.statusCode == 200) {

				const choices = resData.get('choices') as any[];
				const content = ((choices[0] as UTSJSONObject).get('message') as UTSJSONObject).get('content') as string;
				simulationResult.value = cleanMarkdown(content);
			}
		},
		complete: () => { simLoading.value = false; }
	});
};

const copyText = () => { uni.setClipboardData({ data: optimizedText.value }); };
